cmake_minimum_required(VERSION 3.25)
set(CMAKE_SYSTEM_NAME WindowsStore)
set(CMAKE_SYSTEM_VERSION 10.0)
project(wwhd_uwp_python LANGUAGES C CXX)
set(CMAKE_MSVC_RUNTIME_LIBRARY MultiThreadedDLL)
set(WWHD_PYTHON_SOURCE "" CACHE PATH "Verified CPython 3.11.17 source")
set(WWHD_ZLIB_SOURCE "" CACHE PATH "zlib source")
include(sources.cmake)
configure_file("${WWHD_ZLIB_SOURCE}/zconf.h.in" "${CMAKE_CURRENT_BINARY_DIR}/zconf.h" COPYONLY)
add_library(wwhd-python SHARED ${WWHD_PYTHON_SOURCES} config.c uwp_support.c)
target_include_directories(wwhd-python PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}" "${WWHD_PYTHON_SOURCE}/Include" "${WWHD_PYTHON_SOURCE}/Include/internal" "${WWHD_PYTHON_SOURCE}/PC" "${WWHD_ZLIB_SOURCE}" "${CMAKE_CURRENT_BINARY_DIR}")
target_compile_definitions(wwhd-python PRIVATE WIN32 WINVER=0x0A00 _WIN32_WINNT=0x0A00 NTDDI_VERSION=0x0A000008 WINAPI_FAMILY=WINAPI_FAMILY_APP _CRT_APP Py_BUILD_CORE Py_BUILD_CORE_BUILTIN Py_ENABLE_SHARED WWHD_UWP=1 _Py_HAVE_ZLIB MS_DLL_ID="3.11" PY3_DLLNAME=L"wwhd-python.dll")
target_compile_options(wwhd-python PRIVATE /O2 /W2 /we4013 /utf-8)
target_link_options(wwhd-python PRIVATE /APPCONTAINER)
target_link_libraries(wwhd-python PRIVATE WindowsApp bcrypt pathcch version ws2_32)
set_source_files_properties("${WWHD_PYTHON_SOURCE}/Modules/getpath.c" PROPERTIES COMPILE_DEFINITIONS "PREFIX=NULL;EXEC_PREFIX=NULL;VERSION=NULL;VPATH=\"\";PYDEBUGEXT=\"\";PLATLIBDIR=\"Lib\"")
file(GLOB ZLIB_C "${WWHD_ZLIB_SOURCE}/*.c")
list(FILTER ZLIB_C EXCLUDE REGEX "/(example|minigzip)\\.c$")
add_library(wwhd-zlib STATIC ${ZLIB_C})
target_compile_definitions(wwhd-zlib PRIVATE WINAPI_FAMILY=WINAPI_FAMILY_APP _CRT_APP)
target_include_directories(wwhd-zlib PRIVATE "${WWHD_ZLIB_SOURCE}" "${CMAKE_CURRENT_BINARY_DIR}")
target_link_libraries(wwhd-python PRIVATE wwhd-zlib)
set_target_properties(wwhd-python PROPERTIES ARCHIVE_OUTPUT_NAME python311)
set_source_files_properties("${WWHD_PYTHON_SOURCE}/Python/sysmodule.c" PROPERTIES COMPILE_DEFINITIONS "VPATH=\"\"")

add_library(wwhd-python-host STATIC python_host.cpp)
target_include_directories(wwhd-python-host PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}" PRIVATE "${WWHD_PYTHON_SOURCE}/Include" "${WWHD_PYTHON_SOURCE}/PC")
target_compile_features(wwhd-python-host PRIVATE cxx_std_17)
target_compile_definitions(wwhd-python-host PRIVATE WINAPI_FAMILY=WINAPI_FAMILY_APP _CRT_APP)
target_compile_options(wwhd-python-host PRIVATE /EHsc /W4)
target_link_libraries(wwhd-python-host PUBLIC wwhd-python WindowsApp)
