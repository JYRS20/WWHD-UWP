cmake_minimum_required(VERSION 3.25)
project(wwhd_embedded_extract LANGUAGES CXX)
add_library(wwhd-extract-embedded STATIC ../../tools/wudextract/wwhd_extract.cpp ../../tools/wudextract/crypto.cpp)
target_include_directories(wwhd-extract-embedded PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
target_compile_features(wwhd-extract-embedded PUBLIC cxx_std_17)
target_compile_definitions(wwhd-extract-embedded PRIVATE WWHD_EXTRACT_EMBEDDED NOMINMAX)
if(CMAKE_SYSTEM_NAME STREQUAL "WindowsStore")
 target_compile_definitions(wwhd-extract-embedded PRIVATE WWHD_EXTRACT_UWP WINAPI_FAMILY=WINAPI_FAMILY_APP _CRT_APP)
 target_link_libraries(wwhd-extract-embedded PUBLIC WindowsApp)
endif()

option(WWHD_EXTRACT_BUILD_TESTS "Build host-only tests using upstream synthetic fixtures" OFF)
if(WWHD_EXTRACT_BUILD_TESTS)
 if(CMAKE_SYSTEM_NAME STREQUAL "WindowsStore")
  message(FATAL_ERROR "Fixture runner is a host console test, not an Xbox executable")
 endif()
 add_executable(wwhd-extract-embedded-test extract_embedded_test.cpp)
 target_link_libraries(wwhd-extract-embedded-test PRIVATE wwhd-extract-embedded)
endif()
