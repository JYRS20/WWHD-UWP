cmake_minimum_required(VERSION 3.25)
set(CMAKE_SYSTEM_NAME WindowsStore)
set(CMAKE_SYSTEM_VERSION 10.0)
project(wwhd_uwp_game LANGUAGES C CXX)
if(NOT CMAKE_CXX_COMPILER_ID MATCHES "Clang")
 message(FATAL_ERROR "The game runtime requires Clang's musttail support")
endif()
set(CMAKE_C_STANDARD 11)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_MSVC_RUNTIME_LIBRARY MultiThreadedDLL)
get_filename_component(WWHD_ROOT "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
set(WWHD_SDL_SOURCE "" CACHE PATH "Pinned SDL3 UWP source")
set(WWHD_SDL_BUILD "" CACHE PATH "Separately MSVC-built SDL3 UWP directory")
set(WWHD_VULKAN_HEADERS "" CACHE PATH "Pinned Mesa include directory")
option(WWHD_DYNAMIC_GAMECODE "Load the user-compiled game from USB" OFF)
option(WWHD_UNIFIED_PREPARATION "Prepare and launch the game in one application" OFF)
set(WWHD_PYTHON_BUILD "" CACHE PATH "Completed UWP Python build")
if(WWHD_UNIFIED_PREPARATION AND NOT WWHD_DYNAMIC_GAMECODE)
 message(FATAL_ERROR "Unified preparation requires dynamic game code")
endif()
set(WWHD_CLANG_BUILD "" CACHE PATH "Completed UWP Clang build for runtime linking")
set(WWHD_GAMECODE "${WWHD_ROOT}/.local/gamecode-uwp/wwhd-uwp-gamecode.lib" CACHE FILEPATH "Private recompiled game library")
set(GEN_DIR "${WWHD_ROOT}/.local/generated/usa" CACHE PATH "Private generated game headers")
set(WWHD_REQUIRED "${WWHD_SDL_SOURCE}/include/SDL3/SDL.h" "${WWHD_SDL_BUILD}/SDL3.lib" "${WWHD_VULKAN_HEADERS}/vulkan/vulkan.h")
if(NOT WWHD_DYNAMIC_GAMECODE)
 list(APPEND WWHD_REQUIRED "${WWHD_GAMECODE}" "${GEN_DIR}/funcs.h")
endif()
foreach(required IN LISTS WWHD_REQUIRED)
 if(NOT EXISTS "${required}")
  message(FATAL_ERROR "Missing ${required}")
 endif()
endforeach()
add_compile_definitions(WINAPI_FAMILY=WINAPI_FAMILY_APP _CRT_APP WWHD_UWP=1 NOMINMAX _CRT_SECURE_NO_WARNINGS _USE_MATH_DEFINES)
link_directories(BEFORE "$ENV{VCToolsInstallDir}/lib/x64/store")
include(FetchContent)
set(ENABLE_GLSLANG_BINARIES OFF CACHE BOOL "" FORCE)
set(ENABLE_OPT OFF CACHE BOOL "" FORCE)
set(GLSLANG_TESTS OFF CACHE BOOL "" FORCE)
FetchContent_Declare(glslang SOURCE_DIR "${WWHD_ROOT}/.local/host-tools/_deps/glslang-src")
FetchContent_MakeAvailable(glslang)
file(COPY "${glslang_SOURCE_DIR}/SPIRV/" DESTINATION "${CMAKE_BINARY_DIR}/glslang-include/glslang/SPIRV" FILES_MATCHING PATTERN "*.h" PATTERN "*.hpp")
target_include_directories(SPIRV INTERFACE "${CMAKE_BINARY_DIR}/glslang-include")
set(ZLIB_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
FetchContent_Declare(zlib SOURCE_DIR "${WWHD_ROOT}/.local/host-tools/_deps/zlib-src")
FetchContent_MakeAvailable(zlib)
set(LZ4_BUILD_CLI OFF CACHE BOOL "" FORCE)
set(LZ4_BUILD_LEGACY_LZ4C OFF CACHE BOOL "" FORCE)
set(LZ4_BUNDLED_MODE ON)
FetchContent_Declare(lz4 URL https://github.com/lz4/lz4/archive/refs/tags/v1.10.0.tar.gz
 URL_HASH SHA256=537512904744b35e232912055ccf8ec66d768639ff3abe5788d90d792ec5f48b SOURCE_SUBDIR build/cmake DOWNLOAD_EXTRACT_TIMESTAMP TRUE)
FetchContent_MakeAvailable(lz4)
add_library(SDL3_UWP SHARED IMPORTED)
set_target_properties(SDL3_UWP PROPERTIES IMPORTED_IMPLIB "${WWHD_SDL_BUILD}/SDL3.lib" IMPORTED_LOCATION "${WWHD_SDL_BUILD}/SDL3.dll" INTERFACE_INCLUDE_DIRECTORIES "${WWHD_SDL_SOURCE}/include")
if(WWHD_DYNAMIC_GAMECODE)
 add_library(gamecode SHARED IMPORTED)
 set_target_properties(gamecode PROPERTIES IMPORTED_IMPLIB "${WWHD_CLANG_BUILD}/lib/wwhd-clang-bridge.lib" IMPORTED_LOCATION "${WWHD_CLANG_BUILD}/bin/wwhd-clang-bridge.dll")
 add_compile_definitions(WWHD_DYNAMIC_GAMECODE=1 WWHD_EXTRACT_UWP=1)
else()
 add_library(gamecode STATIC IMPORTED)
 set_target_properties(gamecode PROPERTIES IMPORTED_LOCATION "${WWHD_GAMECODE}")
endif()
set(CEMU_DIR "${WWHD_ROOT}/runtime/third_party/cemu")
file(GLOB_RECURSE CEMU_SOURCES CONFIGURE_DEPENDS "${CEMU_DIR}/*.cpp")
list(FILTER CEMU_SOURCES EXCLUDE REGEX "LatteDecompilerEmitMSL[^/]*\\.cpp$")
add_library(cemu_latte STATIC ${CEMU_SOURCES})
target_include_directories(cemu_latte PUBLIC "${CEMU_DIR}" "${CEMU_DIR}/Cafe" "${WWHD_ROOT}/runtime/third_party/fmt/include")
target_compile_options(cemu_latte PRIVATE "/FI${CEMU_DIR}/cemu_shim.h" /W0)
target_compile_definitions(cemu_latte PUBLIC ENABLE_VULKAN=1)
set(IMGUI_DIR "${WWHD_ROOT}/runtime/third_party/imgui")
add_library(imgui STATIC "${IMGUI_DIR}/imgui.cpp" "${IMGUI_DIR}/imgui_draw.cpp" "${IMGUI_DIR}/imgui_tables.cpp" "${IMGUI_DIR}/imgui_widgets.cpp")
target_include_directories(imgui PUBLIC "${IMGUI_DIR}")
target_compile_definitions(imgui PUBLIC IMGUI_DISABLE_DEMO_WINDOWS IMGUI_DISABLE_OBSOLETE_FUNCTIONS)
file(GLOB RUNTIME_SOURCES CONFIGURE_DEPENDS "${WWHD_ROOT}/runtime/src/*.cpp" "${WWHD_ROOT}/runtime/src/*.c" "${WWHD_ROOT}/runtime/src/hle/*.cpp" "${WWHD_ROOT}/runtime/src/mods/*.cpp" "${WWHD_ROOT}/runtime/src/motion/*.cpp" "${WWHD_ROOT}/runtime/src/overlay/*.cpp")
if(WWHD_DYNAMIC_GAMECODE)
 list(FILTER RUNTIME_SOURCES EXCLUDE REGEX "/guest_addr_identity[.]c$")
 file(MAKE_DIRECTORY "${CMAKE_BINARY_DIR}/generated")
 include("${WWHD_ROOT}/cmake/DynamicGameBindings.cmake")
 wwhd_dynamic_bindings("${WWHD_ROOT}" "${CMAKE_BINARY_DIR}/generated/dynamic_bindings.cpp")
 list(APPEND RUNTIME_SOURCES "${CMAKE_BINARY_DIR}/generated/dynamic_bindings.cpp")
endif()
file(GLOB GX2_SOURCES CONFIGURE_DEPENDS "${WWHD_ROOT}/runtime/src/gx2/*.cpp")
file(GLOB VULKAN_SOURCES CONFIGURE_DEPENDS "${WWHD_ROOT}/runtime/src/gfx/vulkan/*.cpp")
set_source_files_properties(${GX2_SOURCES} ${VULKAN_SOURCES} PROPERTIES COMPILE_OPTIONS "/FI${CEMU_DIR}/cemu_shim.h")
# Keep optional CPU instructions isolated; SDL_HasAVX2 gates every call.
set_source_files_properties("${WWHD_ROOT}/runtime/src/gfx/vulkan/uwp_texture_hash_avx2.cpp" PROPERTIES COMPILE_OPTIONS "/clang:-mavx2")
foreach(unit pipeline_recipe pass_cache pipeline_demand shader_translation shader_compiler deferred_commands uwp_driver)
 set_source_files_properties("${WWHD_ROOT}/runtime/src/gfx/vulkan/${unit}.cpp" PROPERTIES COMPILE_OPTIONS "")
endforeach()
set_source_files_properties("${WWHD_ROOT}/runtime/src/button_glyph_font.cpp" "${WWHD_ROOT}/runtime/src/game_glyph.cpp" PROPERTIES COMPILE_OPTIONS "/FI${CEMU_DIR}/cemu_shim.h")
set_source_files_properties("${WWHD_ROOT}/runtime/src/main.cpp" PROPERTIES COMPILE_DEFINITIONS "main=wwhd_game_main")
set(BUILD_INFO "${CMAKE_BINARY_DIR}/generated/build_info.cpp")
add_custom_command(OUTPUT "${BUILD_INFO}" COMMAND ${CMAKE_COMMAND} "-DSRC=${WWHD_ROOT}" "-DOUT=${BUILD_INFO}" -DVERSION= -DVARIANT=uwp -P "${WWHD_ROOT}/cmake/BuildInfo.cmake" VERBATIM)
set(NPC "${CMAKE_BINARY_DIR}/generated/npc_recipes.cpp")
add_custom_command(OUTPUT "${NPC}" COMMAND ${CMAKE_COMMAND} "-DDIR=${WWHD_ROOT}/runtime/assets/npc_catalogue" "-DOUT=${NPC}" -P "${WWHD_ROOT}/cmake/EmbedNpcRecipes.cmake" VERBATIM)
set(BAG "${CMAKE_BINARY_DIR}/generated/story_bag_layout.cpp")
add_custom_command(OUTPUT "${BAG}" COMMAND ${CMAKE_COMMAND} "-DIN=${WWHD_ROOT}/runtime/assets/story_bag/StoryBag_00.json" "-DOUT=${BAG}" -DSYMBOL=wwhd_story_bag_layout -P "${WWHD_ROOT}/cmake/EmbedBytes.cmake" VERBATIM)
add_executable(wwhd-uwp-game WIN32 entry.cpp ${RUNTIME_SOURCES} ${GX2_SOURCES} ${VULKAN_SOURCES}
 "${WWHD_ROOT}/runtime/src/gfx/renderer.cpp" "${WWHD_ROOT}/runtime/src/gfx/display_modes.cpp" "${WWHD_ROOT}/runtime/src/gfx/gamepad_ui_state.cpp"
 "${WWHD_ROOT}/runtime/src/platform/input_sdl.cpp" "${WWHD_ROOT}/runtime/src/platform/mouse_sdl.cpp" "${WWHD_ROOT}/runtime/src/platform/display_rate.cpp"
 "${WWHD_ROOT}/runtime/src/platform/perf_hint.cpp" "${WWHD_ROOT}/runtime/src/platform/thread_sampler.cpp" "${WWHD_ROOT}/runtime/src/platform/touch_pad.cpp" "${WWHD_ROOT}/runtime/src/platform/https_win.cpp" "${BUILD_INFO}" "${NPC}" "${BAG}")
target_include_directories(wwhd-uwp-game PRIVATE "${WWHD_ROOT}/platforms/uwp-clang" "${WWHD_ROOT}/runtime/include" "${WWHD_ROOT}/runtime/src" "${WWHD_ROOT}/runtime/src/gx2" "${WWHD_VULKAN_HEADERS}" "${zlib_SOURCE_DIR}" "${zlib_BINARY_DIR}")
if(NOT WWHD_DYNAMIC_GAMECODE)
 target_include_directories(wwhd-uwp-game PRIVATE "${GEN_DIR}")
endif()
target_compile_definitions(wwhd-uwp-game PRIVATE WWHD_HAS_VULKAN=1 WWHD_SDL_HOST=1 VK_NO_PROTOTYPES WWHD_UWP_EXECUTABLE="wwhd-uwp-game.exe")
target_compile_options(wwhd-uwp-game PRIVATE /EHsc /W2 /clang:-ffp-contract=off)
target_link_options(wwhd-uwp-game PRIVATE /APPCONTAINER /STACK:16777216)
target_link_libraries(wwhd-uwp-game PRIVATE gamecode cemu_latte imgui SDL3_UWP zlibstatic lz4_static glslang::glslang glslang::SPIRV glslang::glslang-default-resource-limits WindowsApp winhttp ws2_32)
if(WWHD_UNIFIED_PREPARATION)
 add_subdirectory(../uwp-extract ${CMAKE_BINARY_DIR}/extract)
 add_library(python-core SHARED IMPORTED)
 set_target_properties(python-core PROPERTIES IMPORTED_IMPLIB "${WWHD_PYTHON_BUILD}/python311.lib" IMPORTED_LOCATION "${WWHD_PYTHON_BUILD}/wwhd-python.dll")
 add_library(python-host STATIC IMPORTED)
 set_target_properties(python-host PROPERTIES IMPORTED_LOCATION "${WWHD_PYTHON_BUILD}/wwhd-python-host.lib")
 target_sources(wwhd-uwp-game PRIVATE preparation.cpp preparation_ui.cpp ../uwp-clang/compile_host.cpp)
 target_compile_definitions(wwhd-uwp-game PRIVATE WWHD_UNIFIED_PREPARATION=1)
 target_link_libraries(wwhd-uwp-game PRIVATE python-host python-core wwhd-extract-embedded)
endif()
if(WWHD_DYNAMIC_GAMECODE)
 target_link_libraries(wwhd-uwp-game PRIVATE bcrypt)
 if(WWHD_UNIFIED_PREPARATION)
  configure_file(UnifiedPackage.appxmanifest.in AppxManifest.xml @ONLY)
 else()
  configure_file(DynamicPackage.appxmanifest.in AppxManifest.xml @ONLY)
 endif()
else()
 configure_file(Package.appxmanifest.in AppxManifest.xml @ONLY)
endif()

include("${WWHD_ROOT}/cmake/PipelineRuntimeIdentity.cmake")
wwhd_pipeline_runtime_identity("${WWHD_ROOT}" WWHD_PIPELINE_RUNTIME_ID)
set_property(SOURCE "${WWHD_ROOT}/runtime/src/gfx/vulkan/backend.cpp" APPEND PROPERTY COMPILE_DEFINITIONS "WWHD_PIPELINE_RUNTIME_ID=\"${WWHD_PIPELINE_RUNTIME_ID}\"")


get_filename_component(LLVM_BIN "${CMAKE_CXX_COMPILER}" DIRECTORY)

 message(FATAL_ERROR "LLVM compiler-rt builtins are required for 128-bit division")
endif()
